    private List<ContratDto> contrats = new();
    private int currentPage = 1;
    private int pageSize = 5;
    private int totalItems = 0;
    private string url = "http://localhost:5077/api/contrat";

    protected override async Task OnInitializedAsync()
    {
        await LoadContrats(1);
    }

    private async Task LoadContrats(int page)
    {
        currentPage = page;
        try
        {
            var options = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };
            var response = await Http.GetFromJsonAsync<ApiResponse>($"{url}?page={currentPage}&pageSize={pageSize}", options);
            if (response != null)
            {
                response.HandleResponse<List<ContratDto>>();
                contrats = (List<ContratDto>)response.Data;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Erreur lors du chargement des contrats: {ex.Message}");
        }
    }

    private void HandleUpdate(int id)
    {
        NavigationManager.NavigateTo($"/contrat/{id}");
    }
